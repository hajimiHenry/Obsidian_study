\documentclass[tikz,border=2bp]{standalone}
\usepackage{ctex}
\usepackage[europeanresistors, americanvoltages, americancurrents, siunitx]{circuitikz}
\standaloneenv{circuitikz}

\begin{document}
\begin{circuitikz}[semithick, scale=1.2, transform shape]
  % 1. 绘制上方电路支路 (C=1F, 关联参考方向)
  % 定义基准 y 轴高度为 4.0
  \def\cy{4.0}
  
  \draw (0.5,\cy) node[left] {a} to[short, o-*] (1.0,\cy)
        to[C, l={$C = \SI{1}{\farad}$}, i^>={$i_C(t)$}, v_>={$u_C(t)$}] (3.0,\cy)
        to[short, *-o] (3.5,\cy) node[right] {b};

  % 2. 绘制下方波形图
  % 绘制坐标轴
  \draw[->] (-0.5,0) -- (5.0,0) node[right] {$t / \text{s}$};
  \draw[->] (0,-0.3) -- (0,2.0) node[above] {$u_C(t) / \text{V}$};
  
  % 绘制刻度和数值
  \draw (0,0) node[below left] {$0$};
  \foreach \x in {1, 2, 4} {
    \draw (\x, 0.05) -- (\x, -0.05) node[below] {$\x$};
  }
  \draw (0.05, 1) -- (-0.05, 1) node[left] {$1$};
  
  % 绘制虚线辅助线
  \draw[dashed, thin, gray] (1,0) -- (1,1) -- (0,1);
  \draw[dashed, thin, gray] (2,0) -- (2,1) -- (1,1);
  
  % 绘制波形折线 (教材手绘蓝线风格)
  \draw[very thick, blue!80!black] (0,0) -- (1,1) -- (2,1) -- (4,0) -- (4.8,0);

\end{circuitikz}
\end{document}
