\documentclass[tikz,border=2bp]{standalone}
\usepackage{ctex}
\usepackage[europeanresistors, americanvoltages, americancurrents, siunitx]{circuitikz}
\usetikzlibrary{arrows.meta}
\standaloneenv{circuitikz}

\begin{document}

  % ==================== (a) 冲激响应电路 ====================
  \begin{circuitikz}[semithick, scale=1.0, transform shape]
    % 串联 RC 电路 (us = δ(t))
    \draw (0,0) to[vsource, l={$\delta(t)$}] (0,2)
          to[R, l={$1\,\Omega$}] (2,2)
          to[C, l={$1\,\text{F}$}] (2,0)
          -- (0,0);
    \draw (2,0) node[ground] {};
    \node[left] at (2.0,1.5) {$+$};
    \node[left] at (2.0,0.5) {$-$};
    \node[left] at (1.9,1.0) {$h(t)$};
    
    \node[below] at (1.0,-0.5) {(a) 冲激响应电路};
  \end{circuitikz}

  % 冲激响应波形 h(t) = e^-t
  \begin{circuitikz}[semithick, scale=1.0, transform shape]
    \draw[-{Stealth}] (-0.2,0) -- (2.5,0) node[right] {$t/\text{s}$};
    \draw[-{Stealth}] (0,-0.2) -- (0,2.2) node[above] {$h(t)/\text{V}$};
    \node[below left] at (0,0) {0};
    
    \draw[dashed, gray] (0,1) -- (2.0,1);
    \draw (-0.05,1) -- (0.05,1) node[left] {1};
    
    % 从 t=0^+ 开始的衰减曲线
    \draw[red, very thick, domain=0:2.2, samples=100] plot (\x, {exp(-\x)});
    \draw[dashed, red, thick] (0,0) -- (0,1); % 0 到 1 的跳变
    
    \node[below] at (1.1,-0.5) {冲激响应 $h(t)$};
  \end{circuitikz}

  % ==================== (b) 矩形脉冲激励 us(t) ====================
  \begin{circuitikz}[semithick, scale=1.0, transform shape]
    \draw[-{Stealth}] (-0.2,0) -- (3.5,0) node[right] {$t/\text{s}$};
    \draw[-{Stealth}] (0,-0.2) -- (0,2.5) node[above] {$u_s(t)/\text{V}$};
    \node[below left] at (0,0) {0};
    
    % 脉冲高 10V (我们用 1.5cm 高度代表 10V)
    \draw[blue, very thick] (0,0) -- (0,1.5) -- (1.5,1.5) -- (1.5,0) -- (3.0,0);
    
    \draw (1.5,0.05) -- (1.5,-0.05) node[below] {2};
    \draw (0.05,1.5) -- (-0.05,1.5) node[left] {10};
    
    \node[below] at (1.65,-0.5) {(b) 矩形脉冲 $u_s(t)$};
  \end{circuitikz}

  % ==================== (c) 卷积响应波形 uC(t) ====================
  \begin{circuitikz}[semithick, scale=1.0, transform shape]
    \draw[-{Stealth}] (-0.2,0) -- (4.5,0) node[right] {$t/\text{s}$};
    \draw[-{Stealth}] (0,-0.2) -- (0,2.5) node[above] {$u_C(t)/\text{V}$};
    \node[below left] at (0,0) {0};
    
    % 在图形中，1.5cm 高度代表 10V，那么 8.65V 对应 1.5 * 0.865 = 1.3cm
    % 0 <= t < 2: uc(t) = 10*(1 - e^-t)
    \draw[red, very thick, domain=0:1.5, samples=100] plot (\x, {1.5*(1 - exp(-\x*1.333))});
    
    % t >= 2 (这里 x 映射：t=2 对应 x=1.5): uc(t) = 8.65 * e^-(t-2)
    \draw[red, very thick, domain=1.5:4.0, samples=100] plot (\x, {1.5*(1 - exp(-2.0)) * exp(-(\x - 1.5)*1.333)});
    
    % 标出 t=2 处的关键点
    \draw[dashed, gray] (1.5,0) -- (1.5, {1.5*(1-exp(-2))}) -- (0, {1.5*(1-exp(-2))});
    \draw (1.5,0.05) -- (1.5,-0.05) node[below] {2};
    \node[left, font=\small] at (0, 1.3) {8.65};
    
    \node[left, font=\small] at (0, 1.5) {10};
    \draw[dashed, gray] (0,1.5) -- (1.5,1.5);
    
    \node[below] at (2.0,-0.5) {(c) 脉冲响应 $u_C(t)$};
  \end{circuitikz}

\end{document}
